# Create a mapping for seasons and weather

# Stacked Bar Plot: weekday vs. casual/registered
weekly_users = df_day.groupby('weekday')[['casual', 'registered']].sum().reset_index()
day_labels = {0: 'Sun', 1: 'Mon', 2: 'Tue', 3: 'Wed', 4: 'Thu', 5: 'Fri', 6: 'Sat'}
